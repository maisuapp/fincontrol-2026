function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Lê os lançamentos da planilha var sheetLanc = ss.getSheetByName("Lançamentos_2026"); var rows = sheetLanc ? sheetLanc.getDataRange().getValues() : []; // 2. Lê a escala do casal var sheetEscala = ss.getSheetByName("Escalas_2026"); var escalaData = ""; var ultimoAutor = ""; var dataAtualizacao = ""; if (sheetEscala && sheetEscala.getLastRow() >= 2) { escalaData = sheetEscala.getRange(2, 1).getValue(); ultimoAutor = sheetEscala.getRange(2, 2).getValue(); dataAtualizacao = sheetEscala.getRange(2, 3).getValue(); } var payload = { rows: rows, escala: escalaData, ultimoAutor: ultimoAutor, dataAtualizacao: dataAtualizacao }; return ContentService.createTextOutput(JSON.stringify(payload)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); // AÇÃO 1: EXCLUIR E ESTORNAR LINHA NA PLANILHA if (data.action === "delete_row") { var sheet = ss.getSheetByName("Lançamentos_2026"); if (sheet) { var values = sheet.getDataRange().getValues(); var targetData = String(data.data || '').split('T')[0].trim(); var targetVal = parseFloat(data.valor) || 0; var targetDesc = String(data.estabelecimento || '').trim(); for (var i = values.length - 1; i >= 1; i--) { var rData = String(values[i][0] || '').split('T')[0].trim(); var rVal = parseFloat(values[i]) || 0; var rDesc = String(values[i] || '').trim(); if ((rData === targetData || rData.includes(targetData)) && Math.abs(rVal - targetVal) < 0.01 && (rDesc === targetDesc || rDesc.includes(targetDesc) || targetDesc.includes(rDesc))) { sheet.deleteRow(i + 1); return ContentService.createTextOutput(JSON.stringify({ status: "success", action: "deleted", row: i + 1 })) .setMimeType(ContentService.MimeType.JSON); } } } return ContentService.createTextOutput(JSON.stringify({ status: "not_found" })) .setMimeType(ContentService.MimeType.JSON); } // AÇÃO 2: SALVAR ESCALA DO CASAL if (data.action === "save_escala") { var sheetEscala = ss.getSheetByName("Escalas_2026"); if (!sheetEscala) { sheetEscala = ss.insertSheet("Escalas_2026"); sheetEscala.appendRow(["JSON_ESCALA", "ULTIMO_AUTOR", "DATA_ATUALIZACAO"]); } var jsonStr = typeof data.todasEscalas === 'string' ? data.todasEscalas : JSON.stringify(data.todasEscalas); sheetEscala.getRange(2, 1).setValue(jsonStr); sheetEscala.getRange(2, 2).setValue(data.updatedBy || "Alguém"); sheetEscala.getRange(2, 3).setValue(new Date().toISOString()); return ContentService.createTextOutput(JSON.stringify({ status: "success", action: "escala" })) .setMimeType(ContentService.MimeType.JSON); } // AÇÃO 3: GRAVAR NOVO LANÇAMENTO (Despesa, Receita ou Reserva) var sheet = ss.getSheetByName("Lançamentos_2026"); sheet.appendRow([data.data, '', data.tipo, data.categoria, data.valor, data.estabelecimento, data.observacoes || '']); return ContentService.createTextOutput(JSON.stringify({ status: "success", action: "append" })) .setMimeType(ContentService.MimeType.JSON); }
